logs: /var/log/vmfb
action buttons and overlay: /etc/motioneye 


When motion event starts:
echo "0" > /etc/motioneye/dep_flag.txt

When video is recorded:
if [ $(cat dep_flag.txt)==1 ]; then 
depfile=$(find /var/lib/motioneye/ -type f -exec ls -t {} + | grep -E '.mp4$' | head -n 1);
cp -T $depfile  "${depfile%%.*}"_DEP."${depfile##*.}"; fi
